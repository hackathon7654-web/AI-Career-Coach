func (c *OrderConsumer) Start(ctx context.Context) error {
	msgs, err := c.channel.Consume(
		c.queueName,
		"order-worker-1",
		false, // manual ack required for delivery guarantees
		false,
		false,
		false,
		nil,
	)
	if err != nil {
		return fmt.Errorf("failed to register amqp consumer: %w", err)
	}

	go func() {
		for {
			select {
			case <-ctx.Done():
				c.logger.Info("stopping consumer loop", zap.Error(ctx.Err()))
				return
			case d, ok := <-msgs:
				if !ok {
					c.logger.Warn("amqp delivery channel closed")
					return
				}
				c.handleDelivery(ctx, d)
			}
		}
	}()
	return nil
}
